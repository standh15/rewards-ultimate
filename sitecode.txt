<script>
const read=(k,d)=>{try{return JSON.parse(localStorage.getItem(k))??d}catch{return d}};
const store=(k,v)=>localStorage.setItem(k,JSON.stringify(v));
let balance=read('ru_balance',0);
let goal=read('ru_goal',0);
let history=read('ru_history',[]);
let done=read('ru_done',[]);
const tasks=['Microsoft Rewards-dashboard bekijken','Bing zoeken','Dagelijkse Rewards-opdracht controleren'];

function safe(s){return String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function render(){
 document.getElementById('balance').textContent=balance.toLocaleString('nl-NL')+' punten';
 document.getElementById('activities').textContent=history.length;
 const month=new Date().getMonth();
 const year=new Date().getFullYear();
 const earned=history.filter(x=>{let d=new Date(x.date);return d.getMonth()===month&&d.getFullYear()===year}).reduce((a,x)=>a+x.points,0);
 document.getElementById('month').textContent=earned.toLocaleString('nl-NL');
 let pct=goal>0?Math.min(100,balance/goal*100):0;
 document.getElementById('bar').style.width=pct+'%';
 document.getElementById('progressText').textContent=goal?balance.toLocaleString('nl-NL')+' / '+goal.toLocaleString('nl-NL')+' punten ('+pct.toFixed(1)+'%)':'Stel een spaardoel in.';
 document.getElementById('history').innerHTML=history.length?history.map(x=>'<li><b>+'+x.points.toLocaleString('nl-NL')+' punten</b><br><small>'+safe(x.reason)+' · '+safe(x.date)+'</small></li>').join(''):'<li>Nog geen activiteiten.</li>';
 document.getElementById('tasks').innerHTML=tasks.map((t,i)=>'<label style="display:block;padding:12px 0"><input style="width:auto" type="checkbox" '+(done.includes(i)?'checked':'')+' onchange="toggleTask('+i+',this.checked)"> '+t+'</label>').join('');
}
function updateBalance(){
 let n=document.getElementById('newBalance').value;
 if(n===''||!Number.isSafeInteger(Number(n))||Number(n)<0)return alert('Vul een geldig saldo in.');
 balance=Number(n);store('ru_balance',balance);render();
 document.getElementById('newBalance').value='';
}
function updateGoal(){
 let n=document.getElementById('goal').value;
 if(n===''||!Number.isSafeInteger(Number(n))||Number(n)<=0)return alert('Vul een geldig doel in.');
 goal=Number(n);store('ru_goal',goal);render();
}
function addPoints(){
 let n=Number(document.getElementById('points').value);
 let r=document.getElementById('reason').value.trim();
 if(!Number.isSafeInteger(n)||n<=0||!r)return alert('Vul een geldig aantal en omschrijving in.');
 if(!Number.isSafeInteger(balance+n))return alert('Saldo is te groot.');
 balance+=n;
 history.unshift({points:n,reason:r,date:new Date().toLocaleDateString('nl-NL')});
 store('ru_balance',balance);store('ru_history',history);
 document.getElementById('points').value='';
 document.getElementById('reason').value='';
 render();
}
function toggleTask(i,v){
 done=v?[...new Set([...done,i])]:done.filter(x=>x!==i);
 store('ru_done',done);
}
function clearHistory(){
 if(confirm('Geschiedenis echt wissen?')){
 history=[];store('ru_history',history);render();
 }
}
render();
